    <script>
        (function() {
            // Armazenamento de dados
            let products = JSON.parse(localStorage.getItem('barProducts')) || [];
            let dailySales = JSON.parse(localStorage.getItem('barDailySales')) || [];
            let accounts = JSON.parse(localStorage.getItem('barAccounts')) || [];
            let currentAccountId = null;

            // Funções de salvamento
            function saveProducts() {
                localStorage.setItem('barProducts', JSON.stringify(products));
            }
            function saveSales() {
                localStorage.setItem('barDailySales', JSON.stringify(dailySales));
            }
            function saveAccounts() {
                localStorage.setItem('barAccounts', JSON.stringify(accounts));
            }

            // Navegação entre tabs
            function switchTab(tabName) {
                document.querySelectorAll('.tabs').forEach(tab => tab.classList.remove('active'));
                document.getElementById(tabName + '-tab').classList.add('active');
                
                document.querySelectorAll('.nav-item').forEach(btn => btn.classList.remove('active'));
                const activeBtn = document.querySelector(`.nav-item[data-tab="${tabName}"]`);
                if (activeBtn) activeBtn.classList.add('active');
                
                renderCurrentTab();
            }

            // Renderização principal
            function renderCurrentTab() {
                renderProducts();
                renderSalesProducts();
                renderDailySales();
                renderAccounts();
            }

            // Cadastro de produtos
            function addProduct() {
                const name = document.getElementById('productName').value.trim();
                const price = parseFloat(document.getElementById('productPrice').value);
                if (name && price > 0) {
                    products.push({ id: Date.now(), name, price });
                    saveProducts();
                    document.getElementById('productName').value = '';
                    document.getElementById('productPrice').value = '';
                    renderProducts();
                } else {
                    alert('Por favor, preencha nome e preço do produto!');
                }
            }

            function deleteProduct(id) {
                if (confirm('Tem certeza que deseja excluir este produto?')) {
                    products = products.filter(p => p.id !== id);
                    saveProducts();
                    renderProducts();
                }
            }

            function renderProducts() {
                const container = document.getElementById('productsList');
                if (products.length === 0) {
                    container.innerHTML = '<div class="empty-state">Nenhum produto cadastrado</div>';
                } else {
                    container.innerHTML = products.map(product => `
                        <div class="sale-item">
                            <div class="sale-info">
                                <strong>${product.name}</strong>
                                <div style="color: #28a745;">R$ ${product.price.toFixed(2)}</div>
                            </div>
                            <button class="btn-delete" data-action="delete-product" data-id="${product.id}">Excluir</button>
                        </div>
                    `).join('');
                }
            }

            // Vendas do dia
            function renderSalesProducts() {
                const container = document.getElementById('salesProducts');
                if (products.length === 0) {
                    container.innerHTML = '<div class="empty-state">Cadastre produtos primeiro</div>';
                } else {
                    container.innerHTML = products.map(product => `
                        <div class="product-card" data-action="register-sale" data-id="${product.id}">
                            <div class="product-name">${product.name}</div>
                            <div class="product-price">R$ ${product.price.toFixed(2)}</div>
                        </div>
                    `).join('');
                }
            }

            function registerSale(productId) {
                const product = products.find(p => p.id === productId);
                if (product) {
                    dailySales.push({
                        id: Date.now(),
                        productName: product.name,
                        price: product.price,
                        date: new Date().toLocaleString('pt-BR'),
                        timestamp: new Date().toISOString()
                    });
                    saveSales();
                    renderDailySales();
                }
            }

            function deleteSale(id) {
                if (confirm('Tem certeza que deseja excluir esta venda?')) {
                    dailySales = dailySales.filter(s => s.id !== id);
                    saveSales();
                    renderDailySales();
                }
            }

            function renderDailySales() {
                const container = document.getElementById('dailySales');
                const totalContainer = document.getElementById('totalSales');
                const today = new Date().toDateString();
                const todaySales = dailySales.filter(sale => new Date(sale.timestamp).toDateString() === today);
                
                if (todaySales.length === 0) {
                    container.innerHTML = '<div class="empty-state">Nenhuma venda hoje</div>';
                } else {
                    container.innerHTML = todaySales.map(sale => `
                        <div class="sale-item">
                            <div class="sale-info">
                                <strong>${sale.productName}</strong>
                                <div>R$ ${sale.price.toFixed(2)}</div>
                                <div class="sale-time">${sale.date}</div>
                            </div>
                            <button class="btn-delete" data-action="delete-sale" data-id="${sale.id}">Excluir</button>
                        </div>
                    `).join('');
                }
                const total = todaySales.reduce((sum, sale) => sum + sale.price, 0);
                totalContainer.innerHTML = `Total do dia: R$ ${total.toFixed(2)}`;
            }

            // Contas
            function createAccount() {
                const name = document.getElementById('accountName').value.trim();
                if (name) {
                    accounts.push({
                        id: Date.now(),
                        name,
                        items: [],
                        status: 'open',
                        date: new Date().toISOString()
                    });
                    saveAccounts();
                    document.getElementById('accountName').value = '';
                    renderAccounts();
                } else {
                    alert('Por favor, digite o nome da conta!');
                }
            }

            function selectAccount(accountId) {
                currentAccountId = accountId;
                document.getElementById('accountsProducts').style.display = 'block';
                renderAccountProducts();
                document.getElementById('accountsProducts').scrollIntoView({ behavior: 'smooth' });
            }

            function addToAccount(productId) {
                if (!currentAccountId) return;
                const product = products.find(p => p.id === productId);
                const account = accounts.find(a => a.id === currentAccountId);
                if (product && account && account.status === 'open') {
                    account.items.push({
                        id: Date.now(),
                        productName: product.name,
                        price: product.price,
                        date: new Date().toLocaleString('pt-BR'),
                        paid: false
                    });
                    saveAccounts();
                    renderAccounts();
                    renderAccountProducts();
                }
            }

            function deleteAccountItem(accountId, itemId) {
                const account = accounts.find(a => a.id === accountId);
                if (account) {
                    account.items = account.items.filter(i => i.id !== itemId);
                    saveAccounts();
                    renderAccounts();
                }
            }

            function toggleItemPayment(accountId, itemId) {
                const account = accounts.find(a => a.id === accountId);
                if (account) {
                    const item = account.items.find(i => i.id === itemId);
                    if (item) {
                        item.paid = !item.paid;
                        saveAccounts();
                        renderAccounts();
                    }
                }
            }

            function closeAccount(accountId) {
                if (confirm('Tem certeza que deseja fechar esta conta?')) {
                    const account = accounts.find(a => a.id === accountId);
                    if (account) {
                        account.status = 'closed';
                        saveAccounts();
                        renderAccounts();
                    }
                }
            }

            function checkAccount(accountId) {
                const account = accounts.find(a => a.id === accountId);
                if (account) {
                    account.status = 'checked';
                    saveAccounts();
                    renderAccounts();
                }
            }

            function uncheckAccount(accountId) {
                const account = accounts.find(a => a.id === accountId);
                if (account) {
                    account.status = 'open';
                    saveAccounts();
                    renderAccounts();
                }
            }

            function renderAccountProducts() {
                const container = document.getElementById('accountProductsGrid');
                if (products.length === 0) {
                    container.innerHTML = '<div class="empty-state">Cadastre produtos primeiro</div>';
                } else {
                    container.innerHTML = products.map(product => `
                        <div class="product-card" data-action="add-to-account" data-id="${product.id}">
                            <div class="product-name">${product.name}</div>
                            <div class="product-price">R$ ${product.price.toFixed(2)}</div>
                        </div>
                    `).join('');
                }
            }

            function renderAccounts() {
                const container = document.getElementById('accountsList');
                if (accounts.length === 0) {
                    container.innerHTML = '<div class="empty-state">Nenhuma conta criada</div>';
                } else {
                    container.innerHTML = accounts.map(account => {
                        const totalValue = account.items.reduce((sum, item) => sum + item.price, 0);
                        const unpaidValue = account.items.filter(i => !i.paid).reduce((sum, item) => sum + item.price, 0);
                        const totalPaid = account.items.filter(i => i.paid).length;
                        
                        let statusClass = '';
                        if (account.status === 'closed') statusClass = 'closed';
                        else if (account.status === 'checked') statusClass = 'checked';
                        
                        return `
                            <div class="account-item ${statusClass}">
                                <div class="account-header">
                                    <span class="account-name">${account.name}</span>
                                    <span class="account-total">R$ ${totalValue.toFixed(2)}</span>
                                </div>
                                <small>Status: ${account.status === 'open' ? 'Aberta' : account.status === 'checked' ? 'Marcada' : 'Fechada'}</small>
                                <small> | Pago: ${totalPaid}/${account.items.length} itens</small>
                                ${unpaidValue > 0 ? `<div style="color: #dc3545;">A pagar: R$ ${unpaidValue.toFixed(2)}</div>` : ''}
                                
                                ${account.items.length > 0 ? `
                                    <div class="account-items">
                                        ${account.items.map(item => `
                                            <div style="display: flex; justify-content: space-between; align-items: center; padding: 5px 0; ${item.paid ? 'text-decoration: line-through; color: #666;' : ''}">
                                                <span>${item.productName} - R$ ${item.price.toFixed(2)}</span>
                                                <div>
                                                    ${account.status !== 'closed' ? `
                                                        <button data-action="toggle-payment" data-account="${account.id}" data-item="${item.id}" class="${item.paid ? 'btn-pay' : ''}" style="padding: 2px 5px; font-size: 11px; margin: 0 2px;">${item.paid ? '✓ Pago' : 'Pagar'}</button>
                                                        <button data-action="delete-account-item" data-account="${account.id}" data-item="${item.id}" class="btn-delete" style="padding: 2px 5px; font-size: 11px;">Excluir</button>
                                                    ` : ''}
                                                </div>
                                            </div>
                                        `).join('')}
                                    </div>
                                ` : '<div style="color: #666; margin: 10px 0;">Nenhum item adicionado</div>'}
                                
                                <div class="btn-group">
                                    ${account.status === 'open' ? `
                                        <button data-action="select-account" data-id="${account.id}">Adicionar Produtos</button>
                                        <button data-action="check-account" data-id="${account.id}" class="btn-check">Marcar</button>
                                        <button data-action="close-account" data-id="${account.id}" class="btn-close">Fechar</button>
                                    ` : account.status === 'checked' ? `
                                        <button data-action="select-account" data-id="${account.id}">Adicionar Produtos</button>
                                        <button data-action="uncheck-account" data-id="${account.id}">Desmarcar</button>
                                        <button data-action="close-account" data-id="${account.id}" class="btn-close">Fechar</button>
                                    ` : `
                                        <div style="color: #dc3545; font-weight: bold;">Conta Fechada</div>
                                    `}
                                </div>
                            </div>
                        `;
                    }).join('');
                }
            }

            // Limpar vendas antigas
            function cleanOldSales() {
                const today = new Date().toDateString();
                dailySales = dailySales.filter(sale => new Date(sale.timestamp).toDateString() === today);
                saveSales();
            }

            // Delegação de eventos para evitar onclick inline
            document.addEventListener('click', function(e) {
                const target = e.target;
                
                // Navegação inferior
                if (target.closest('.nav-item')) {
                    const tab = target.closest('.nav-item').getAttribute('data-tab');
                    if (tab) switchTab(tab);
                }
                
                // Botão Adicionar Produto
                if (target.id === 'btnAddProduct' || target.closest('#btnAddProduct')) {
                    addProduct();
                }
                
                // Botão Nova Conta
                if (target.id === 'btnCreateAccount' || target.closest('#btnCreateAccount')) {
                    createAccount();
                }
                
                // Ações por data attributes
                const action = target.getAttribute('data-action') || target.closest('[data-action]')?.getAttribute('data-action');
                if (!action) return;
                
                const id = target.getAttribute('data-id') || target.closest('[data-action]')?.getAttribute('data-id');
                const accountId = target.getAttribute('data-account') || target.closest('[data-action]')?.getAttribute('data-account');
                const itemId = target.getAttribute('data-item') || target.closest('[data-action]')?.getAttribute('data-item');
                
                switch(action) {
                    case 'delete-product':
                        deleteProduct(parseInt(id));
                        break;
                    case 'register-sale':
                        registerSale(parseInt(id));
                        break;
                    case 'delete-sale':
                        deleteSale(parseInt(id));
                        break;
                    case 'add-to-account':
                        addToAccount(parseInt(id));
                        break;
                    case 'select-account':
                        selectAccount(parseInt(id));
                        break;
                    case 'check-account':
                        checkAccount(parseInt(id));
                        break;
                    case 'uncheck-account':
                        uncheckAccount(parseInt(id));
                        break;
                    case 'close-account':
                        closeAccount(parseInt(id));
                        break;
                    case 'toggle-payment':
                        toggleItemPayment(parseInt(accountId), parseInt(itemId));
                        break;
                    case 'delete-account-item':
                        deleteAccountItem(parseInt(accountId), parseInt(itemId));
                        break;
                }
            });

            // Inicialização
            cleanOldSales();
            renderCurrentTab();
        })();
    </script>
